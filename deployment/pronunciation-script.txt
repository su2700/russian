 const spelling = word => word.normalize('NFC').toLowerCase().replaceAll('́','');
 function lookup(word, written=word) {
  const key=spelling(word);
  if(!/^[а-яё]+$/u.test(key))return;
  if(Object.hasOwn(exceptions,key)) {
   const [display,reason]=exceptions[key];
   return {display,reason};
  }
  if(key.length<2||!Array.from(key).some(letter=>vowels.includes(letter)))return;
  const printed=written.normalize('NFC');
  const knownStress=stress.get(key);
  const display=printed.includes('́')||!knownStress?printed:
   /^[А-ЯЁ]/u.test(printed)?knownStress[0].toUpperCase()+knownStress.slice(1):knownStress;
  const lower=display.toLowerCase();
  let stressed=lower.indexOf('́')-1;
  if(stressed<0)stressed=lower.indexOf('ё');
  const changes=[];
  const letters=spelling(display);
  if(stressed>=0&&letters===key){
   const reductions=[];
   for(let index=0;index<letters.length;index++){
    if(index===stressed)continue;
    const letter=letters[index],previous=letters[index-1];
    if((letter==='о'||letter==='а')&&(index===0||hard.includes(previous))){
     // Loanwords can retain unstressed о next to a vowel, such as ра́дио.
     if(letter==='о'&&vowels.includes(letters[index+1]||' '))continue;
     const near=index<stressed&&!Array.from(letters.slice(index+1,stressed)).some(c=>vowels.includes(c));
     const sound=near||index===0?'[ɐ]':'[ə]';
     if(previous&&'жшц'.includes(previous)&&letter==='а'&&near)continue;
     reductions.push((index+1)+' 个字母 '+letter+' → '+sound);
    } else if((letter==='е'||letter==='я')&&previous&&'чщ'.includes(previous)&&index<stressed){
     reductions.push((index+1)+' 个字母 '+letter+' → 近似 [ɪ]');
    }
   }
   if(reductions.length)changes.push('按标出的重音轻读非重读元音：第 '+reductions.join('；第 ')+'。');
  }
  if(/ть?ся$/u.test(key)){
   changes.push('动词词尾 -тся / -ться 连成 [t͡sːə]，像稍长的 ц 加一个很轻的元音。');
  }
  const separator=key.match(/([ьъ])([еёюя])/u);
  if(separator){
   const [,sign,letter]=separator;
   changes.push(sign+' 本身无声，在 '+letter+' 前起分隔作用，让 '+letter+' 保留 [j] 开头。');
  } else if(/[^жшчщ]ь$/u.test(key)&&!/ть?ся$/u.test(key)){
   changes.push('词尾的 ь 本身无声，前面的 '+key.at(-2)+' 读软音。');
  } else if(/[жшчщ]ь$/u.test(key)){
   changes.push('词尾的 ь 不发音；前面的 '+key.at(-2)+' 保持原来的软硬。');
  }
  if(/[жшц]и/u.test(key))changes.push('ж、ш、ц 保持硬音，紧跟的 и 听起来接近 [ɨ]（ы 音）。');
  const initialGlide=key.match(/^[еёюя]|[аеёиоуыэюя][еёюя]/u);
  if(initialGlide&&!separator)changes.push(initialGlide[0].at(-1)+' 在词首或元音后有 [j] 开头，后面的元音仍要看重音。');
  for(let index=0;index<key.length-1;index++){
   const letter=key[index],next=key[index+1];
   if(voiced[letter]&&'пфктшсхцчщ'.includes(next)&&!(letter==='ж'&&next==='ч')){
    changes.push(letter+' 在清辅音 '+next+' 前清化，接近 ['+voiced[letter]+']。');
   }else if(beforeVoiced[letter]&&'бгджз'.includes(next)){
    changes.push(letter+' 在浊辅音 '+next+' 前浊化，接近 ['+beforeVoiced[letter]+']。');
   }
  }
  const final=key.at(-1);
  if(voiced[final])changes.push('单独读或停顿前，词尾 '+final+' 清化为 ['+voiced[final]+']。');
  if(/[дт]ц/u.test(key))changes.push('дц / тц 连读时，直接收进 ц 的塞擦音，通常不把前一个辅音单独读出来。');
  if(!changes.length)return;
  return {display,reason:changes.join(' ')};
 }
 function append(cell,note) {

  const details=document.createElement('details');
  details.className='p11-note';
  const summary=document.createElement('summary');

  details.append(summary);
  for(const text of [note.reason,note.tip].filter(Boolean)){
   const paragraph=document.createElement('p');
   paragraph.textContent=text;
   details.append(paragraph);
  }
  cell.append(details);
 }

 }
